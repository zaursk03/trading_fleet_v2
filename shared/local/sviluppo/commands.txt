docker compose run --rm srv-dev download-data --config /local/sviluppo/config-dev.json --timeframes 15m 5m 1h
# probe 1m (retention): una coppia sola, non alla cieca
docker compose run --rm srv-dev download-data --config /local/sviluppo/config-dev.json --pairs BTC/USDC:USDC --timeframes 1m --timerange 20261001-

# NB: futures mode scarica anche funding 1h + mark. NON disabilitare ftcache.

# ── 2 · coverage (script rimossi): controllo inline ────────────
docker compose run --rm --entrypoint python srv-dev -c "import pandas as pd,glob; fs=glob.glob('/freqtrade/user_data/data/hyperliquid/futures/*-15m-futures.feather'); print('coppie 15m:',len(fs)); df=pd.read_feather(fs[0]); print('esempio:',df['date'].min(),df['date'].max(),len(df))"

# ── 3 · backtesting (strategy da experiment.json; detail 1m se presente, else 5m=ottimista) ──
docker compose run --rm srv-dev backtesting --config /local/sviluppo/config-dev.json --timerange 20240101-20260801 --timeframe-detail 1m
# fallback 5m (risultati ottimistici → dichiararlo nel report)
docker compose run --rm srv-dev backtesting --config /local/sviluppo/config-dev.json --timerange 20240101-20260801 --timeframe-detail 5m

# ── 4 · bias check PRIMA di fidarsi del backtest ───────────────
docker compose run --rm srv-dev lookahead-analysis --config /local/sviluppo/config-dev.json --timerange 20240101-20260801
docker compose run --rm srv-dev recursive-analysis --config /local/sviluppo/config-dev.json --timerange 20240101-20260801

# ── 5 · hyperopt (200-300 epochs; PlateauSampler; delete .json co-located prima; no log in strategy) ──
docker compose run --rm srv-dev hyperopt --config /local/sviluppo/config-dev.json --hyperopt-loss MoutonMeanRevHyperOptLoss --sampler PlateauSampler --epochs 250 --spaces buy sell roi stoploss --timerange 20240101-20260101
# NB: niente trailing nativo; loss Calmar/MoutonMeanRev per mean-reversion

# ── 6 · walk-forward CPCV + PBO (<0.5) + holdout intoccabile ───
# conferma i nomi dei flag wf-* con --help (fork-specific):
docker compose run --rm srv-dev walk-forward --help
docker compose run --rm srv-dev walk-forward --config /local/sviluppo/config-dev.json --hyperopt-loss MoutonMeanRevHyperOptLoss --sampler PlateauSampler --epochs 250 --wf-mode cpcv --wf-cpcv-groups 6 --wf-cpcv-test-groups 2 --wf-embargo-days 3 --wf-holdout-months 2 --timerange 20240101-20260801

# ── 7 · replay (parity live-vs-backtest; MAI per selezionare parametri) ──

# ── 9 · fleet ops (bot2 LIVE: mai compose run; logs/backup/kill-switch) ──
docker logs --tail 100 bot2
# backup a caldo (WAL-safe). sqlite3 CLI è presente nell'immagine: alternativa sqlite3 … ".backup …"
docker compose exec -T srv-bot2 python -c "import sqlite3;s=sqlite3.connect('file:/freqtrade/user_data/tradesv3.sqlite?mode=ro',uri=True);d=sqlite3.connect('/freqtrade/user_data/tradesv3-backup.sqlite');s.backup(d);d.close();s.close();print('backup ok')"
mv bot2/user_data/tradesv3-backup.sqlite backups/bot2-$(date +%Y%m%d).sqlite
find backups -name 'bot2-*.sqlite' -mtime +7 -delete
